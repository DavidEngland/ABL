## 1. Core Shared Library (phys/module_most_general_lib.F)  
To maintain exact physical consistency between surface exchange coefficients ($C_M, C_H$) and column diffusivities ($K_m, K_h$), encapsulate the analytical solver and integrated profile functions into a single, pure Fortran module.  
  
For linear flux-profile relations $\phi_m(\zeta) = 1 + \beta_m \zeta$ and $\phi_h(\zeta) = 1 + \beta_h \zeta$, the integrated Monin-Obukhov stability functions are:  
  
$$\psi_m(\zeta) = \int_0^\zeta \frac{1 - \phi_m(\zeta')}{\zeta'} d\zeta' = -\beta_m \zeta, \quad \psi_h(\zeta) = \int_0^\zeta \frac{1 - \phi_h(\zeta')}{\zeta'} d\zeta' = -\beta_h \zeta$$  
Fortran  
  
MODULE module_most_general_lib  
   IMPLICIT NONE  
  
CONTAINS  
  
   !-------------------------------------------------------------------  
   ! Inverts Ri_g to compute zeta, phi_m, phi_h, fm, fh, Pr_t.  
   ! Uses analytical branch limit Ri_lim = beta_h / (beta_m**2).  
   !-------------------------------------------------------------------  
   ELEMENTAL PURE SUBROUTINE evaluate_most_general( &  
        Ri_g, beta_m, beta_h, &  
        zeta, phi_m, phi_h, fm, fh, pr_t &  
   )  
      REAL, INTENT(IN)  :: Ri_g, beta_m, beta_h  
      REAL, INTENT(OUT) :: zeta, phi_m, phi_h, fm, fh, pr_t  
        
      REAL :: Ri_lim, disc, denom  
  
      Ri_lim = beta_h / (beta_m**2)  
  
      IF (Ri_g >= Ri_lim .OR. Ri_g < 0.0) THEN  
         zeta  = 0.0  
         phi_m = 1.0  
         phi_h = 1.0  
         fm    = 0.0  
         fh    = 0.0  
         pr_t  = 1.0  
         RETURN  
      END IF  
  
      disc  = MAX(0.0, 1.0 + 4.0 * (beta_h - beta_m) * Ri_g)  
      denom = (1.0 - 2.0 * beta_m * Ri_g) + SQRT(disc)  
  
      IF (denom <= 0.0) THEN  
         zeta  = 0.0  
         phi_m = 1.0  
         phi_h = 1.0  
         fm    = 0.0  
         fh    = 0.0  
         pr_t  = 1.0  
      ELSE  
         zeta  = (2.0 * Ri_g) / denom  
         phi_m = 1.0 + beta_m * zeta  
         phi_h = 1.0 + beta_h * zeta  
         fm    = 1.0 / (phi_m**2)  
         fh    = 1.0 / (phi_m * phi_h)  
         pr_t  = phi_h / phi_m  
      END IF  
   END SUBROUTINE evaluate_most_general  
  
   !-------------------------------------------------------------------  
   ! Integrated profile functions psi_m(zeta) and psi_h(zeta)  
   !-------------------------------------------------------------------  
   ELEMENTAL PURE SUBROUTINE evaluate_psi_general( &  
        zeta, beta_m, beta_h, psi_m, psi_h &  
   )  
      REAL, INTENT(IN)  :: zeta, beta_m, beta_h  
      REAL, INTENT(OUT) :: psi_m, psi_h  
  
      psi_m = -beta_m * zeta  
      psi_h = -beta_h * zeta  
   END SUBROUTINE evaluate_psi_general  
  
END MODULE module_most_general_lib  
## 2. Surface Layer Integration (phys/module_sf_custom.F)  
The custom Surface Layer scheme computes friction velocity $u_*$, temperature scale $\theta_*$, Monin-Obukhov length $L$, and exchange coefficients $C_M, C_H$ at the first grid level $z_1$. It invokes module_most_general_libto guarantee identical physics at the air-sea/land interface.  
  
Fortran  
  
MODULE module_sf_custom  
   USE module_most_general_lib, ONLY: evaluate_psi_general  
   USE module_model_constants,  ONLY: karman, g  
  
CONTAINS  
  
   SUBROUTINE custom_surface_layer( &  
        u1, v1, th1, th_sfc, z1, &  
        beta_m, beta_h, &  
        ustar, qstar, cm, ch, mol &  
   )  
      IMPLICIT NONE  
  
      REAL, INTENT(IN)  :: u1, v1, th1, th_sfc, z1, beta_m, beta_h  
      REAL, INTENT(OUT) :: ustar, qstar, cm, ch, mol  
  
      INTEGER, PARAMETER :: max_iter = 10  
      REAL, PARAMETER    :: z0 = 0.1, z0h = 0.1, tol = 1.0e-4  
      REAL               :: V_mag, dth, L_old, zeta, psi_m, psi_h  
      INTEGER            :: iter  
  
      V_mag = MAX(SQRT(u1**2 + v1**2), 0.1)  
      dth   = th1 - th_sfc  
      mol   = 100.0 ! Initial seed for L  
  
      ! Iterate Monin-Obukhov length L = u_*^2 * theta_0 / (g * karman * theta_*)  
      DO iter = 1, max_iter  
         L_old = mol  
         zeta  = MAX(0.0, z1 / mol)  
  
         CALL evaluate_psi_general(zeta, beta_m, beta_h, psi_m, psi_h)  
  
         ustar = (karman * V_mag) / (LOG(z1 / z0) - psi_m)  
         qstar = (karman * dth)   / (LOG(z1 / z0h) - psi_h)  
  
         IF (ABS(qstar) > 1.0e-6) THEN  
            mol = (ustar**2 * th_sfc) / (g * karman * qstar)  
         ELSE  
            mol = 1.0e5  
         END IF  
  
         IF (ABS(mol - L_old) / ABS(L_old) < tol) EXIT  
      END DO  
  
      cm = (ustar / V_mag)**2  
      ch = (ustar * qstar) / (V_mag * dth)  
   END SUBROUTINE custom_surface_layer  
  
END MODULE module_sf_custom

## 4. Staged Verification Workflow  
Stage 1: Julia Forward Verification  
 └── Generate synthetic (Ri_g) profiles across [0, Ri_lim).  
 └── Verify explicit inverse zeta(Ri_g) yields exact input Ri_g.  
  
Stage 2: WRF SCM (ideal/scm_xy)  
 └── Run single-column cases (GABLS1, SHEBA).  
 └── Validate Pr_t convergence and K_m, K_h flux profiles against Julia reference output.  
  
Stage 3: Idealized 3D Boundary Layer (ideal/em_les or ideal/em_scm_xy)  
 └── Evaluate topological suppression factor C(d_fold) = tanh(d_fold / delta_0).  
 └── Confirm transport anisotropy invariance (K_m / K_h = Pr_t).  
  
Stage 4: Mesoscale Comparison  
 └── Compare against MYNN Level 2.5 and YSU schemes under stable nocturnal conditions.  
To run Stage 2, build WRF's Single Column Model executable:  
Bash  
  
./compile scm_xy  
cd test/em_scm_xy  
./ideal.exe  
./wrf.exe
